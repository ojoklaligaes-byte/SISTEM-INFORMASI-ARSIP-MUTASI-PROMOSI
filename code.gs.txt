/**
 * Automatically creates required sheets and headers if they don't exist
 */
function ensureDatabaseStructure() {
  const cache = CacheService.getScriptCache();
  const VER = 'v15'; // naikkan nilai ini setiap menambah kolom/sheet baru di ensureDatabaseStructure_
  if (cache.get('db_ok_' + VER)) return; // struktur sudah dicek (hemat waktu di tiap panggilan)
  const props = PropertiesService.getScriptProperties();
  // penanda permanen: pemeriksaan struktur yang berat hanya dijalankan sekali per versi kode
  if (props.getProperty('DB_STRUCT_OK') === VER) { cache.put('db_ok_' + VER, '1', 21600); return; }
  ensureDatabaseStructure_();
  props.setProperty('DB_STRUCT_OK', VER);
  cache.put('db_ok_' + VER, '1', 21600);
}

/**
 * Memulihkan master baku K01-K40: hanya menambahkan ID yang hilang, data yang ada tidak diubah (Admin)
 */
function restoreMasterBaku(token, samakan) {
  requireAuth(token, 'Admin');
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.MASTER_KEGIATAN);
    const data = sheet.getDataRange().getValues();
    const have = {};
    for (let i = 1; i < data.length; i++) have[String(data[i][0]).trim()] = true;
    let fixed = 0;
    if (samakan) { // samakan nama/urutan/status K01-K40 yang sudah ada dengan daftar baku
      const std = {};
      getMasterBakuList_().forEach(r => std[r[0]] = r);
      for (let i = 1; i < data.length; i++) {
        const r = std[String(data[i][0]).trim()];
        if (r && (String(data[i][1]).trim() !== r[1] || Number(data[i][2]) !== r[2] || data[i][3] !== r[3])) {
          sheet.getRange(i + 1, 2, 1, 3).setValues([[r[1], r[2], r[3]]]);
          fixed++;
        }
      }
    }
    const missing = getMasterBakuList_().filter(r => !have[r[0]]);
    if (missing.length) sheet.getRange(sheet.getLastRow() + 1, 1, missing.length, 4).setValues(missing);
    CacheService.getScriptCache().remove('master_cache');
    return { success: true, restored: missing.length, fixed: fixed, message: (missing.length ? missing.length + ' jenis kegiatan baku ditambahkan. ' : 'Tidak ada ID yang hilang. ') + (samakan ? fixed + ' nama/urutan disamakan dengan daftar baku.' : '') };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Validates user credentials for login. Mengembalikan token sesi (6 jam).
 */
function loginUser(username, password) {
  const cache = CacheService.getScriptCache();
  const failKey = 'fail_' + String(username).trim().toLowerCase();
  const fails = Number(cache.get(failKey) || 0);
  if (fails >= 5) {
    return { success: false, message: 'Terlalu banyak percobaan gagal. Coba lagi dalam 10 menit.' };
  }

  const data = getUsersRows_();

  for (let i = 1; i < data.length; i++) {
    const [userID, uName, fullname, pwd, role, allowedActivities, status] = data[i];
    if (String(uName).trim() === String(username).trim() && passwordMatches(pwd, password)) {
      if (status !== 'Aktif') {
        return { success: false, message: 'Akun Anda sedang nonaktif. Silakan hubungi Administrator.' };
      }
      // upgrade password lama (plaintext) menjadi hash
      if (String(pwd).indexOf('sha256:') !== 0) {
        SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.USERS).getRange(i + 1, 4).setValue(hashPassword(password));
        bustUsers_();
      }
      cache.remove(failKey);
      const token = Utilities.getUuid() + Utilities.getUuid();
      cache.put('sess_' + token, JSON.stringify({ userID: userID }), 21600);
      return {
        success: true,
        token: token,
        user: { userID: userID, username: uName, fullname: fullname, role: role, allowedActivities: allowedActivities, status: status }
      };
    }
  }
  cache.put(failKey, String(fails + 1), 600);
  return { success: false, message: 'Username atau Password tidak valid!' };
}

function logoutUser(token) {
  if (token) CacheService.getScriptCache().remove('sess_' + token);
  return { success: true };
}

/**
 * Memvalidasi token sesi dan peran. Peran/status selalu dibaca ulang dari sheet.
 * Melempar error SESSION_EXPIRED / AKSES_DITOLAK.
 */
function requireAuth(token, requiredRole) {
  const expired = new Error('SESSION_EXPIRED: Sesi berakhir, silakan login kembali.');
  if (!token) throw expired;
  const cache = CacheService.getScriptCache();
  const raw = cache.get('sess_' + token);
  if (!raw) throw expired;
  const sess = JSON.parse(raw);

  const ukey = 'usr_' + sess.userID;
  const uhit = cache.get(ukey);
  if (uhit) {
    const u = JSON.parse(uhit);
    if (requiredRole === 'Admin' && u.role !== 'Admin') throw new Error('AKSES_DITOLAK: Hanya Admin yang dapat melakukan aksi ini.');
    return u;
  }

  const data = getUsersRows_();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(sess.userID)) {
      if (data[i][6] !== 'Aktif') throw new Error('SESSION_EXPIRED: Akun nonaktif.');
      const user = {
        userID: data[i][0], username: data[i][1], fullname: data[i][2],
        role: data[i][4], allowedActivities: data[i][5]
      };
      cache.put(ukey, JSON.stringify(user), 30); // profil & peran di-cache 30 detik (hemat baca sheet)
      if (requiredRole === 'Admin' && user.role !== 'Admin') {
        throw new Error('AKSES_DITOLAK: Hanya Admin yang dapat melakukan aksi ini.');
      }
      cache.put('sess_' + token, raw, 21600); // perpanjang sesi
      return user;
    }
  }
  throw expired;
}

function isActivityAllowed(user, jenisKegiatanID) {
  if (user.role === 'Admin') return true;
  const allowed = String(user.allowedActivities || '');
  if (allowed === 'ALL') return true;
  return allowed.split(',').map(function (s) { return s.trim(); }).indexOf(String(jenisKegiatanID)) !== -1;
}

function getMasterJenisKegiatan(token) {

}

function getMaster_() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('master_cache');
  if (hit) { try { return JSON.parse(hit); } catch (e) {} }
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.MASTER_KEGIATAN);
  let data = sheet.getDataRange().getValues();
  if (data.length <= 1) { seedMasterJenisKegiatan(sheet); data = sheet.getDataRange().getValues(); } // master kosong -> pulihkan K01-K40
  // Perbaiki otomatis: nama kosong / berisi kode (mis. 'K01') diganti nama baku
  const std = {};
  getMasterBakuList_().forEach(r => std[r[0]] = r[1]);
  let healed = false;
  for (let i = 1; i < data.length; i++) {
    const id = String(data[i][0]).trim();
    const nm = String(data[i][1]).trim();
    if (std[id] && (!nm || /^K\d{1,3}$/i.test(nm))) {
      sheet.getRange(i + 1, 2).setValue(std[id]);
      data[i][1] = std[id];
      healed = true;
    }
  }
  if (healed) CacheService.getScriptCache().remove('master_cache');
  const result = [];

  for (let i = 1; i < data.length; i++) {
    result.push({
      jenisKegiatanID: data[i][0],
      namaJenisKegiatan: data[i][1],
      urutan: Number(data[i][2]),
      status: data[i][3]
    });
  }

  const sorted = result.sort((a, b) => a.urutan - b.urutan);
  try { cache.put('master_cache', JSON.stringify(sorted), 21600); } catch (e) {}
  return sorted;
}

/**
 * Imports multiple Pegawai items from Excel/CSV upload
 */
function importPegawaiBatch_impl_(token, pegawaiList) {
  const auth = requireAuth(token, 'Admin');

  pegawaiList = sanitizeList_(pegawaiList);
  if (!Array.isArray(pegawaiList) || pegawaiList.length === 0) {
    return { success: false, message: 'Tidak ada data pegawai yang valid untuk diimpor.' };
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.PEGAWAI);
    const data = sheet.getDataRange().getValues();
    const digits = function (v) { return String(v || '').replace(/\D/g, ''); };

    // indeks data lama berdasarkan NIP / NIPPPK (upsert: data lama tidak hilang)
    const idx = {};
    for (let i = 1; i < data.length; i++) {
      const k1 = digits(data[i][5]), k2 = digits(data[i][6]);
      if (k1.length >= 8) idx[k1] = i;
      if (k2.length >= 8) idx[k2] = i;
    }

    const fields = ['statusPegawai', 'nama', 'gelarDepan', 'gelarBelakang', 'nip', 'nipppk', 'pangkat', 'golongan', 'jabatan', 'pendidikan', 'alamatRumah', 'unor', 'unitInduk'];
    const now = new Date();
    const dirty = {};
    const toAppend = [];
    let updated = 0, added = 0, skipped = 0;
    let counter = data.length;

    pegawaiList.forEach(function (p) {
      if (!p || !String(p.nama || '').trim()) { skipped++; return; }
      const k1 = digits(p.nip), k2 = digits(p.nipppk);
      const hit = (k1.length >= 8 && idx[k1] !== undefined) ? idx[k1] : ((k2.length >= 8 && idx[k2] !== undefined) ? idx[k2] : -1);

      if (hit > 0) {
        // perbarui hanya kolom yang terisi pada file; kolom kosong tidak menimpa data lama
        fields.forEach(function (k, n) {
          const v = p[k];
          if (v !== undefined && v !== null && String(v).trim() !== '') data[hit][n + 1] = v;
        });
        data[hit][15] = now;
        if (dirty[hit] !== 'new') dirty[hit] = true;
        updated++;
      } else {
        counter++;
        const row = [
          'PEG-IMP-' + now.getTime() + '-' + counter, p.statusPegawai || 'PNS', p.nama, p.gelarDepan || '', p.gelarBelakang || '',
          p.nip || '', p.nipppk || '', p.pangkat || '', p.golongan || '', p.jabatan || '',
          p.pendidikan || '', p.alamatRumah || '', p.unor || '', p.unitInduk || '',
          'IMPORT', now
        ];
        toAppend.push(row);
        // indeks agar duplikat dalam file yang sama menjadi pembaruan, bukan baris ganda
        data.push(row);
        dirty[data.length - 1] = 'new';
        if (k1.length >= 8) idx[k1] = data.length - 1;
        if (k2.length >= 8) idx[k2] = data.length - 1;
        added++;
      }
    });

    // tulis pembaruan baris lama (per blok baris berurutan agar cepat)
    const rows = Object.keys(dirty).map(Number).filter(function (r) { return dirty[r] !== 'new'; }).sort(function (a, b) { return a - b; });
    let start = 0;
    while (start < rows.length) {
      let end = start;
      while (end + 1 < rows.length && rows[end + 1] === rows[end] + 1) end++;
      const block = [];
      for (let r = rows[start]; r <= rows[end]; r++) block.push(data[r].slice(0, 16));
      sheet.getRange(rows[start] + 1, 1, block.length, 16).setValues(block);
      start = end + 1;
    }
    // baris baru: hanya yang benar-benar baru dan masih ada di akhir data (urutan sama)
    const newRows = [];
    for (let r = 1; r < data.length; r++) if (dirty[r] === 'new') newRows.push(data[r].slice(0, 16));
    if (newRows.length) sheet.getRange(sheet.getLastRow() + 1, 1, newRows.length, 16).setValues(newRows);

    return {
      success: true, updated: updated, added: added, skipped: skipped,
      message: 'Impor selesai: ' + added + ' pegawai baru ditambahkan, ' + updated + ' data lama diperbarui (data lama tidak dihapus)' +
        (skipped ? ', ' + skipped + ' baris dilewati (nama kosong).' : '.')
    };
  } finally {
    lock.releaseLock();
  }
}

function storeLampiran_(dataUrl, kegiatanID) {
  const head = /^data:(application\/pdf|image\/png|image\/jpeg);base64,/.exec(String(dataUrl).substring(0, 60));
  if (!head) throw new Error('Format lampiran tidak didukung (hanya PDF, PNG, JPG).');
  const bytes = Utilities.base64Decode(String(dataUrl).substring(head[0].length));
  if (bytes.length > 1048576) throw new Error('Ukuran lampiran melebihi 1 MB.');
  const ext = head[1] === 'application/pdf' ? 'pdf' : (head[1] === 'image/png' ? 'png' : 'jpg');
  const blob = Utilities.newBlob(bytes, head[1], 'SK_' + kegiatanID + '.' + ext);
  return 'drive:' + getLampiranFolder_().createFile(blob).getId();
}

function trashLampiran_(ref) {
  if (ref && String(ref).indexOf('drive:') === 0) {
    try { DriveApp.getFileById(String(ref).substring(6)).setTrashed(true); } catch (e) {}
  }
}

/**
 * Mengambil lampiran (data URL) untuk satu kegiatan. Hanya file yang tercatat di sheet.
 */
function readLampiran_(ref) {
  try {
    const ck = 'lamp_' + ref.substring(6);
    const hit = cacheBigGet_(ck);
    if (hit && hit.dataUrl) return hit;
    const blob = DriveApp.getFileById(ref.substring(6)).getBlob();
    const out = {
      success: true,
      name: blob.getName(),
      dataUrl: 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes())
    };
    if (out.dataUrl.length < 1400000) cacheBigPut_(ck, out, 21600);
    return out;
  } catch (e) {
    return { success: false, message: 'Gagal membaca berkas dari Drive: ' + e.message };
  }
}

function getLampiran(token, kegiatanID, ref) {
  requireAuth(token);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.KEGIATAN);
  ref = String(ref || '');
  // jalur cepat: hanya membaca satu kolom (bukan seluruh sheet)
  if (ref.indexOf('drive:') === 0) {
    const last = sheet.getLastRow();
    if (last >= 2) {
      const f = sheet.getRange(2, 23, last - 1, 1).createTextFinder(ref).matchEntireCell(true).findNext();
      if (f && String(sheet.getRange(f.getRow(), 1).getValue()) === String(kegiatanID)) return readLampiran_(ref);
    }
  }
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(kegiatanID)) {
      const r = String(data[i][22] || '');
      if (r.indexOf('drive:') !== 0) return { success: false, message: 'Tidak ada lampiran.' };
      return readLampiran_(r);
    }
  }
  return { success: false, message: 'Data kegiatan tidak ditemukan.' };
}

  const auth = requireAuth(token);

  const rawFile = String((payload && payload.fileData) || '');
  payload = sanitizeObj_(payload);
  payload.fileData = rawFile; // data URL lampiran jangan diubah
  if (!isActivityAllowed(auth, payload.jenisKegiatanID)) {
    throw new Error('AKSES_DITOLAK: Anda tidak memiliki hak akses untuk jenis kegiatan ini.');
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
  const incoming = String(payload.fileData || '');

  if (payload.kegiatanID) {
    // jalur cepat: cari baris lewat kolom ID, baca hanya satu baris (bukan seluruh sheet)
    const lastRow = sheet.getLastRow();
    let rowNo = -1;
    if (lastRow >= 2) {
      const f = sheet.getRange(2, 1, lastRow - 1, 1).createTextFinder(String(payload.kegiatanID)).matchEntireCell(true).findNext();
      if (f) rowNo = f.getRow();
    }
    if (rowNo > 0) {
      const old = sheet.getRange(rowNo, 1, 1, Math.min(34, sheet.getMaxColumns())).getValues()[0];
      while (old.length < 34) old.push('');
      if (!isActivityAllowed(auth, old[5])) {
        throw new Error('AKSES_DITOLAK: Anda tidak memiliki hak akses untuk mengubah data ini.');
      }
      const izinRow = (auth.role === 'Admin') ? 0 : findIzinAktif_(auth, 'EDIT_KEGIATAN', payload.kegiatanID);
      if (auth.role !== 'Admin' && !izinRow) {
        throw new Error('AKSES_DITOLAK: Mengubah data kegiatan memerlukan izin Admin. Ajukan permintaan izin terlebih dahulu.');
      }
      let fileRef = old[22] || '';
      const rwChanges = diffKegiatan_(old, payload, incoming.indexOf('data:') === 0);
      if (incoming.indexOf('data:') === 0) {
        const newRef = storeLampiran_(incoming, payload.kegiatanID);
        trashLampiran_(fileRef);
        fileRef = newRef;
      }
      // satu kali tulis untuk kolom 2..34 (kolom createdAt & pencatat awal dipertahankan)

        payload.pegawaiID, payload.nama, payload.statusPegawai, payload.nip,

        payload.nomorSK, payload.tanggalSK, payload.tmt, fileRef, 1, payload.keterangan,
        old[25], new Date(),
        payload.tugasTambahanLama || '', payload.tugasTambahanBaru || '', payload.kelasJabatanLama || '', payload.pangkatGolBaru || '',
        old[31], old[32], payload.jumlahKGB || ''
      ]]);
      if (izinRow) pakaiIzin_(izinRow);
      logPerubahan_(auth, 'EDIT', old, rwChanges);
      syncPegawaiDariKegiatan_(auth, payload, payload.kegiatanID);
      return { success: true, message: 'Data kegiatan berhasil diperbarui.' };
    }
  }

  // Insert New Activity Record
  const now = new Date();
  const dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyyMMdd');
  const newID = 'KEG-' + dateStr + '-' + now.getTime();
  const fileRef = incoming.indexOf('data:') === 0 ? storeLampiran_(incoming, newID) : '';

  sheet.appendRow([
    newID, payload.pegawaiID, payload.nama, payload.statusPegawai, payload.nip,
    payload.jenisKegiatanID, payload.namaJenisKegiatan, payload.tanggalKegiatan,
    payload.pangkatGol, payload.jenisJabatan, payload.kelasJabatan,
    payload.jabatanLama, payload.jabatanBaru, payload.unitKerjaLama, payload.unitKerjaBaru,
    payload.instansiAsal, payload.instansiTujuan, payload.nomorUsulan, payload.tanggalUsulan,
    payload.nomorSK, payload.tanggalSK, payload.tmt, fileRef, 1, payload.keterangan, now, '',
    payload.tugasTambahanLama || '', payload.tugasTambahanBaru || '', payload.kelasJabatanLama || '', payload.pangkatGolBaru || '',
    auth.username || '', auth.fullname || '', payload.jumlahKGB || ''

  return { success: true, message: 'Data kegiatan kepegawaian berhasil dicatat.' };

const IZIN_TIPE = ['EDIT_KEGIATAN', 'HAPUS_KEGIATAN', 'EDIT_PEGAWAI', 'HAPUS_PEGAWAI'];

function readIzin_() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.IZIN);
  const data = sheet.getDataRange().getValues();
  const out = [];
  for (let i = 1; i < data.length; i++) {
    out.push({
      row: i + 1,
      requestID: data[i][0], tipe: data[i][1], targetID: data[i][2], targetLabel: data[i][3],
      userID: data[i][4], username: data[i][5], alasan: data[i][6], status: data[i][7],
      createdAt: formatDateTimeString(data[i][8]),
      decidedAt: formatDateTimeString(data[i][9]),
      decidedBy: data[i][10], catatan: data[i][11],
      _decidedRaw: data[i][9]
    });
  }
  return out;
}

function izinKedaluwarsa_(rec) {
  if (rec.status !== 'DISETUJUI') return false;
  const t = rec._decidedRaw instanceof Date ? rec._decidedRaw.getTime() : new Date(rec._decidedRaw).getTime();
  return !t || (Date.now() - t) > IZIN_BERLAKU_JAM * 3600 * 1000;
}

/** Mengembalikan nomor baris izin DISETUJUI (belum dipakai & belum kedaluwarsa) atau 0 */
function findIzinAktif_(auth, tipe, targetID) {
  const list = readIzin_();
  for (let i = list.length - 1; i >= 0; i--) {
    const r = list[i];
    if (r.status === 'DISETUJUI' && r.tipe === tipe && String(r.targetID) === String(targetID) &&
        String(r.userID) === String(auth.userID) && !izinKedaluwarsa_(r)) return r.row;
  }
  return 0;
}

function pakaiIzin_(row) {
  SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.IZIN).getRange(row, 8).setValue('DIGUNAKAN');
}

function getIzin_(auth) {
  const list = readIzin_();
  const out = [];
  list.forEach(function (r) {
    if (auth.role !== 'Admin' && String(r.userID) !== String(auth.userID)) return;
    const status = izinKedaluwarsa_(r) ? 'KEDALUWARSA' : r.status;
    out.push({
      requestID: r.requestID, tipe: r.tipe, targetID: r.targetID, targetLabel: r.targetLabel,
      userID: r.userID, username: r.username, alasan: r.alasan, status: status,
      createdAt: r.createdAt, decidedAt: r.decidedAt, decidedBy: r.decidedBy, catatan: r.catatan
    });
  });
  return out.reverse().slice(0, 300);
}

  return getIzin_(auth);
}

function ajukanIzin(token, tipe, targetID, targetLabel, alasan) {
  const auth = requireAuth(token);
  if (auth.role === 'Admin') return { success: false, message: 'Admin dapat langsung mengubah/menghapus data.' };
  if (IZIN_TIPE.indexOf(tipe) === -1 || !targetID) return { success: false, message: 'Permintaan izin tidak valid.' };
  alasan = String(alasan || '').trim().substring(0, 500);
  if (!alasan) return { success: false, message: 'Alasan permintaan wajib diisi.' };

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const list = readIzin_();
    for (let i = 0; i < list.length; i++) {
      const r = list[i];
      if (String(r.userID) === String(auth.userID) && r.tipe === tipe && String(r.targetID) === String(targetID) &&
          (r.status === 'MENUNGGU' || (r.status === 'DISETUJUI' && !izinKedaluwarsa_(r)))) {
        return { success: true, message: r.status === 'MENUNGGU' ? 'Permintaan sebelumnya masih menunggu persetujuan Admin.' : 'Izin sudah disetujui, silakan lanjutkan.', existing: true };
      }
    }
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.IZIN);
    sheet.appendRow([
      'IZN-' + new Date().getTime(), tipe, String(targetID), sanitizeCell_(String(targetLabel || '').substring(0, 200)),
      auth.userID, auth.username, sanitizeCell_(alasan), 'MENUNGGU', new Date(), '', '', ''
    ]);
    return { success: true, message: 'Permintaan izin dikirim. Menunggu persetujuan Admin.' };
  } finally {
    lock.releaseLock();
  }
}

function putusanIzin(token, requestID, keputusan, catatan) {
  const auth = requireAuth(token, 'Admin');
  if (keputusan !== 'DISETUJUI' && keputusan !== 'DITOLAK') return { success: false, message: 'Keputusan tidak valid.' };
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const list = readIzin_();

      if (String(list[i].requestID) === String(requestID)) {
        if (list[i].status !== 'MENUNGGU') return { success: false, message: 'Permintaan ini sudah diputuskan (' + list[i].status + ').' };
        const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.IZIN);

    }
    return { success: false, message: 'Permintaan tidak ditemukan.' };
  } finally {
    lock.releaseLock();
  }
}

function getDashboardSummary_() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('dash_cache');
  if (hit) { try { return JSON.parse(hit); } catch (e) {} }
  ensureDatabaseStructure();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const peg = ss.getSheetByName(SHEET_NAMES.PEGAWAI);
  const keg = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
  const counts = { total: 0, PNS: 0, CPNS: 0, PPPK: 0, PW: 0 };
  if (peg.getLastRow() > 1) {
    peg.getRange(2, 2, peg.getLastRow() - 1, 1).getValues().forEach(r => { counts.total++; const st = String(r[0]).trim(); counts[st] = (counts[st] || 0) + 1; });
  }
  const perJenis = {};
  const recent = [];
  let total = 0;
  const last = keg.getLastRow();
  if (last > 1) {
    const n = last - 1;
    // hanya 2 kolom (ID jenis & nama jenis) untuk semua baris + 5 baris terakhir -> jauh lebih ringan
    const jn = keg.getRange(2, 6, n, 2).getValues();
    const masterName = {};
    getMaster_().forEach(m => masterName[String(m.jenisKegiatanID).trim()] = m.namaJenisKegiatan);
    const fix = r => { let nm = String(r[1]).trim(); if (!nm || /^K\d{1,3}$/i.test(nm)) nm = masterName[String(r[0]).trim()] || nm; return nm; };
    jn.forEach(r => { total++; const nm = fix(r); if (nm) perJenis[nm] = (perJenis[nm] || 0) + 1; });
    const k = Math.min(5, n);
    const tail = keg.getRange(last - k + 1, 1, k, 8).getValues();
    for (let i = tail.length - 1; i >= 0; i--) {
      const r = tail[i];
      recent.push({ namaJenisKegiatan: fix([r[5], r[6]]), nama: r[2], nip: r[4], tanggalKegiatan: r[7] ? formatDateString(r[7]) : '' });
    }
  }
  const out = { success: true, counts: counts, totalKegiatan: total, perJenis: perJenis, recent: recent };
  try { cache.put('dash_cache', JSON.stringify(out), 600); } catch (e) {}
  return out;
}

/**
 * Saves or updates a user account (Admin saja)
 */
function saveUser_impl_(token, payload) {
  requireAuth(token, 'Admin');
  if (payload && payload.userID) CacheService.getScriptCache().remove('usr_' + payload.userID);
  const rawPassword = payload.password;
  payload = sanitizeObj_(payload);
  payload.password = rawPassword; // password di-hash apa adanya
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.USERS);
    const data = sheet.getDataRange().getValues();
    const uname = String(payload.username).trim();

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][1]).trim().toLowerCase() === uname.toLowerCase() && String(data[i][0]) !== String(payload.userID)) {
        return { success: false, message: 'Username sudah dipakai pengguna lain.' };
      }
    }

    if (payload.userID) {
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(payload.userID)) {
          const wasActiveAdmin = data[i][4] === 'Admin' && data[i][6] === 'Aktif';
          const stillActiveAdmin = payload.role === 'Admin' && payload.status === 'Aktif';
          if (wasActiveAdmin && !stillActiveAdmin && countActiveAdmins_(data) <= 1) {
            return { success: false, message: 'Tidak dapat menonaktifkan/menurunkan Admin aktif terakhir.' };
          }
          sheet.getRange(i + 1, 2).setValue(uname);
          sheet.getRange(i + 1, 3).setValue(payload.fullname);
          if (payload.password) sheet.getRange(i + 1, 4).setValue(hashPassword(payload.password));
          sheet.getRange(i + 1, 5).setValue(payload.role);
          sheet.getRange(i + 1, 6).setValue(payload.allowedActivities);
          sheet.getRange(i + 1, 7).setValue(payload.status);
          return { success: true, message: 'Data akun pengguna berhasil diperbarui.' };
        }
      }
    }

    const newID = 'USR-' + new Date().getTime();
    sheet.appendRow([
      newID, uname, payload.fullname, hashPassword(payload.password || 'Demo2026!'),
      payload.role, payload.allowedActivities, payload.status || 'Aktif', new Date()
    ]);
    return { success: true, message: 'Pengguna baru berhasil ditambahkan.' };
  } finally {
    lock.releaseLock();
  }
}

function countActiveAdmins_(data) {
  let n = 0;
  for (let i = 1; i < data.length; i++) {
    if (data[i][4] === 'Admin' && data[i][6] === 'Aktif') n++;
  }
  return n;
}

    sheet.appendRow([payload.namaPejabat, payload.nipPejabat, payload.jabatanPejabat, payload.instansi, payload.kota, payload.pangkatGol || '']);
  }

  return { success: true, message: 'Pejabat penandatangan laporan berhasil diperbarui.' };
}

/**
 * Updates logged-in user password/username (akun diambil dari token, bukan dari kiriman klien)
 */
function updateAccount_impl_(token, payload) {
  const auth = requireAuth(token);
  CacheService.getScriptCache().remove('usr_' + auth.userID);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.USERS);
  const data = sheet.getDataRange().getValues();
  const newName = sanitizeCell_(String(payload.newUsername || '').trim());
  if (!newName) return { success: false, message: 'Username wajib diisi.' };

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][1]).trim().toLowerCase() === newName.toLowerCase() && String(data[i][0]) !== String(auth.userID)) {
      return { success: false, message: 'Username sudah dipakai pengguna lain.' };
    }
  }

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(auth.userID)) {
      if (payload.newPassword) {
        if (!passwordMatches(data[i][3], payload.oldPassword)) {
          return { success: false, message: 'Password lama tidak cocok!' };
        }
        sheet.getRange(i + 1, 4).setValue(hashPassword(payload.newPassword));
      }
      sheet.getRange(i + 1, 2).setValue(newName);
      return { success: true, newUsername: newName, message: 'Pengaturan akun berhasil diperbarui.' };
    }
  }
  return { success: false, message: 'Akun tidak ditemukan.' };
}

/**
 * Helper: tanggal + jam (WIB) -> 'yyyy-MM-dd HH:mm:ss'
 */
function formatDateTimeString(val) {
  if (!val) return '';
  if (val instanceof Date) {
    return Utilities.formatDate(val, 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss');
  }
  return String(val);
}

/**
 * Helper to convert Date object or string to YYYY-MM-DD
 */
function formatDateString(val) {
  if (!val) return '';
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(val).substring(0, 10);
}

/**
 * Password helpers (SHA-256). Password lama (plaintext) tetap bisa login dan
 * otomatis valid sampai diganti.
 */
function hashPassword(plain) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(plain), Utilities.Charset.UTF_8);
  return 'sha256:' + bytes.map(function (b) {
    return ('0' + (b < 0 ? b + 256 : b).toString(16)).slice(-2);
  }).join('');
}

function passwordMatches(stored, input) {
  stored = String(stored).trim();
  input = String(input).trim();
  if (stored.indexOf('sha256:') === 0) return stored === hashPassword(input);
  return stored === input;
}

/**
 * Mencegah formula injection di Google Sheets: teks yang diawali = + - @
 * diberi awalan apostrof agar disimpan sebagai teks biasa.
 */
function sanitizeCell_(v) {
  if (typeof v === 'string' && /^[=+\-@]/.test(v)) return "'" + v;
  return v;
}

function sanitizeObj_(o) {
  if (!o || typeof o !== 'object') return o;
  const out = {};
  Object.keys(o).forEach(function (k) { out[k] = sanitizeCell_(o[k]); });
  return out;
}

function sanitizeList_(list) {
  return Array.isArray(list) ? list.map(sanitizeObj_) : list;
}

// ===== Cache daftar besar (dipotong < 100KB per kunci) =====
function cacheBigPut_(key, obj, ttl) {
  try {
    const json = JSON.stringify(obj);
    const size = 30000;
    const n = Math.ceil(json.length / size);
    const map = {};
    for (let i = 0; i < n; i++) map[key + '_' + i] = json.substr(i * size, size);
    map[key + '_n'] = String(n);
    CacheService.getScriptCache().putAll(map, ttl);
  } catch (e) {}
}

function cacheBigGet_(key) {
  try {
    const cache = CacheService.getScriptCache();
    const n = Number(cache.get(key + '_n') || 0);
    if (!n) return null;
    const keys = [];
    for (let i = 0; i < n; i++) keys.push(key + '_' + i);
    const m = cache.getAll(keys);
    let json = '';
    for (let i = 0; i < n; i++) {
      const part = m[key + '_' + i];
      if (part === undefined || part === null) return null;
      json += part;
    }
    return JSON.parse(json);
  } catch (e) { return null; }
}

function cacheBigRemove_(key) {
  try {
    const cache = CacheService.getScriptCache();
    const n = Number(cache.get(key + '_n') || 0);
    const keys = [key + '_n'];
    for (let i = 0; i < n; i++) keys.push(key + '_' + i);
    cache.removeAll(keys);
  } catch (e) {}
}

function bustData_() {

  cacheBigRemove_('d_keg');
  bustDash_();
}

function hapusRiwayatPerubahan(token, logID) {
  requireAuth(token, 'Admin');
  ensureDatabaseStructure();
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.RIWAYAT);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const last = sh.getLastRow();
    if (last < 2) return { success: true, message: 'Tidak ada catatan.' };
    if (logID === 'ALL') {
      sh.deleteRows(2, last - 1);
      return { success: true, message: 'Seluruh catatan riwayat dihapus.' };
    }
    const ids = sh.getRange(2, 1, last - 1, 1).getValues();
    for (let i = ids.length - 1; i >= 0; i--) {
      if (String(ids[i][0]) === String(logID)) { sh.deleteRow(i + 2); return { success: true, message: 'Catatan riwayat dihapus.' }; }
    }
    return { success: false, message: 'Catatan tidak ditemukan.' };
  } finally { lock.releaseLock(); }
}


// ---------- SINKRON DATABASE PEGAWAI DARI RIWAYAT KEGIATAN ----------
function pangkatText_(row) {
  const pk = String(row[7] || '').trim(), gl = String(row[8] || '').trim();
  const st = String(row[1] || '').toUpperCase();
  if ((st === 'PNS' || st === 'CPNS') && pk && gl && pk.indexOf('(') === -1 && pk.toUpperCase().indexOf(gl.toUpperCase()) === -1) return pk + ' (' + gl + ')';
  return pk || gl;
}

function logRiwayatJabatan_(auth, row, kegID, namaJenis, jenis, jabSebelum, unorSebelum, indukSebelum, jabSesudah, unorSesudah, tmt, extra) {
  try {
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.RIWAYAT_JABATAN);
    if (!sh) return;
    extra = extra || {};
    const pkNow = pangkatText_(row);
    sh.appendRow(['RJ-' + new Date().getTime() + '-' + Math.floor(Math.random() * 1000), new Date(), row[0], sanitizeCell_(String(row[2] || '')), kegID, namaJenis, jenis,

      sanitizeCell_(String(extra.pkSeb !== undefined ? extra.pkSeb : pkNow)), sanitizeCell_(String(extra.pkSes !== undefined ? extra.pkSes : pkNow)), extra.raw || '']);
  } catch (e) {}
}

/**
 * Saat kegiatan dihapus: kembalikan pangkat, jabatan, unor pegawai ke kondisi sebelum kegiatan tersebut
 * (hanya bila nilai pegawai saat ini masih hasil dari kegiatan itu; perubahan yang lebih baru tidak ditimpa).
 */
function revertPegawaiDariKegiatan_(auth, kegID) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const rj = ss.getSheetByName(SHEET_NAMES.RIWAYAT_JABATAN);
    const sp = ss.getSheetByName(SHEET_NAMES.PEGAWAI);
    const last = rj.getLastRow();
    if (last < 2) return;
    const logs = rj.getRange(2, 1, last - 1, Math.max(rj.getLastColumn(), 17)).getValues();
    for (let k = logs.length - 1; k >= 0; k--) {
      const r = logs[k];
      if (String(r[4]) !== String(kegID)) continue;
      const jenis = String(r[6]);
      if (jenis.indexOf('DIKEMBALIKAN') !== -1) continue;
      const pd = sp.getDataRange().getValues();
      let hit = -1;
      for (let i = 1; i < pd.length; i++) { if (String(pd[i][0]) === String(r[2])) { hit = i; break; } }
      if (hit < 0) continue;
      const row = pd[hit];
      while (row.length < 22) row.push('');
      const rowNo = hit + 1;
      if (jenis === 'DEFINITIF') {
        if (String(row[9]) === String(r[10])) {
          logRiwayatJabatan_(auth, row, kegID, 'Kegiatan dihapus - data dikembalikan', 'DEFINITIF (DIKEMBALIKAN)', row[9], row[12], row[13], r[7], r[8], '');
          sp.getRange(rowNo, 10).setValue(r[7]);
          sp.getRange(rowNo, 13, 1, 2).setValues([[r[8], r[9]]]);
          sp.getRange(rowNo, 21).setValue('');
          sp.getRange(rowNo, 16).setValue(new Date());
        }
      } else if (jenis === 'PANGKAT') {
        if (pangkatText_(row) === String(r[15])) {
          let raw = null;
          try { raw = JSON.parse(r[16] || 'null'); } catch (e) {}
          if (raw) {
            logRiwayatJabatan_(auth, row, kegID, 'Kegiatan dihapus - data dikembalikan', 'PANGKAT (DIKEMBALIKAN)', '', '', '', '', '', '', { pkSeb: pangkatText_(row), pkSes: String(r[14]) });
            sp.getRange(rowNo, 8, 1, 2).setValues([[raw.p, raw.g]]);
            sp.getRange(rowNo, 16).setValue(new Date());
          }
        }
      } else if (jenis === 'PLT' || jenis === 'PLH') {
        if (String(r[10]) && String(row[21]) === String(kegID)) {
          logRiwayatJabatan_(auth, row, kegID, 'Kegiatan dihapus - data dikembalikan', jenis + ' (DIKEMBALIKAN)', row[16], row[17], '', r[7], r[8], '');
          sp.getRange(rowNo, 17, 1, 3).setValues([[r[7], r[8], r[7] ? jenis : '']]);
          sp.getRange(rowNo, 20).setValue('');
          sp.getRange(rowNo, 22).setValue('');
          sp.getRange(rowNo, 16).setValue(new Date());
        }
      }
    }
  } catch (e) {}
}

/**
 * Memulihkan data kegiatan yang sudah dihapus dari Riwayat Perubahan (khusus Admin).
 */
function pulihkanKegiatan(token, logID) {
  const auth = requireAuth(token, 'Admin');
  ensureDatabaseStructure();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const rw = ss.getSheetByName(SHEET_NAMES.RIWAYAT);
    const last = rw.getLastRow();
    if (last < 2) return { success: false, message: 'Catatan tidak ditemukan.' };
    const ids = rw.getRange(2, 1, last - 1, 1).getValues();
    let rowNo = -1;
    for (let i = ids.length - 1; i >= 0; i--) { if (String(ids[i][0]) === String(logID)) { rowNo = i + 2; break; } }
    if (rowNo < 0) return { success: false, message: 'Catatan tidak ditemukan.' };
    const rec = rw.getRange(rowNo, 1, 1, 12).getValues()[0];
    if (String(rec[2]) !== 'HAPUS' || !rec[10]) return { success: false, message: 'Catatan ini tidak memiliki cadangan data untuk dipulihkan.' };
    if (rec[11]) return { success: false, message: 'Data ini sudah dipulihkan sebelumnya.' };
    let snap;
    try { snap = JSON.parse(rec[10]); } catch (e) { return { success: false, message: 'Cadangan data rusak.' }; }
    const sk = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
    const lk = sk.getLastRow();
    if (lk >= 2) {
      const f = sk.getRange(2, 1, lk - 1, 1).createTextFinder(String(snap[0])).matchEntireCell(true).findNext();
      if (f) return { success: false, message: 'Data dengan ID yang sama sudah ada kembali.' };
    }
    while (snap.length < 34) snap.push('');
    [25, 26].forEach(function (ci) { if (snap[ci]) { const d = new Date(snap[ci]); if (!isNaN(d.getTime())) snap[ci] = d; } });
    snap[26] = new Date();
    if (snap[22] && String(snap[22]).indexOf('drive:') === 0) {
      try { DriveApp.getFileById(String(snap[22]).substring(6)).setTrashed(false); } catch (e) { snap[22] = ''; }
    }
    sk.appendRow(snap);
    rw.getRange(rowNo, 12).setValue(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm') + ' oleh ' + auth.username);
    syncPegawaiDariKegiatan_(auth, {
      pegawaiID: snap[1], nip: snap[4], namaJenisKegiatan: snap[6], jabatanBaru: snap[12], unitKerjaBaru: snap[14],
      tmt: snap[21] ? formatDateString(snap[21]) : '', pangkatGolBaru: snap[30] || ''
    }, snap[0]);
    bustData_();
    return { success: true, message: 'Data kegiatan berhasil dipulihkan.' };
  } finally { lock.releaseLock(); }
}

/**
 * Memperbarui DatabasePegawai dari satu kegiatan:
 *  - Jabatan definitif (kolom jabatan/unor/unitInduk) bila kegiatan bukan PLT/PLH & bukan tugas tambahan murni.
 *  - Jabatan PLT/PLH (kolom 17-22) bila jenis kegiatan "Pelaksana Tugas".
 * Nilai sebelumnya selalu disalin ke sheet RiwayatJabatanPegawai (data lama tetap tersimpan).
 * Hanya kegiatan dengan TMT >= TMT terakhir yang diterapkan (edit data lama tidak menimpa yang lebih baru).
 */
function syncPegawaiDariKegiatan_(auth, p, kegID) {
  try {
    const nm = String(p.namaJenisKegiatan || '').toLowerCase();
    const isPlt = nm.indexOf('pelaksana tugas') !== -1;
    const isTT = !isPlt && (nm.indexOf('tugas tambahan') !== -1 || nm.indexOf('kepala sekolah') !== -1 || nm.indexOf('kepala puskesmas') !== -1);
    if (isTT) return;
    const pkBaru = String(p.pangkatGolBaru || '').trim();
    if (pkBaru && /^kenaikan pangkat/i.test(String(p.namaJenisKegiatan || '').trim())) {
      try {
        const shP = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.PEGAWAI);
        const dP = shP.getDataRange().getValues();
        for (let i = 1; i < dP.length; i++) {
          if (p.pegawaiID && String(dP[i][0]) === String(p.pegawaiID)) {
            const m = /\(([^)]+)\)/.exec(pkBaru);
            const rowP = dP[i];
            const pkSebText = pangkatText_(rowP);
            logRiwayatJabatan_(auth, rowP, kegID, p.namaJenisKegiatan, 'PANGKAT', '', '', '', '', '', String(p.tmt || '').substring(0, 10), { pkSeb: pkSebText, pkSes: pkBaru, raw: JSON.stringify({ p: rowP[7], g: rowP[8] }) });
            shP.getRange(i + 1, 8, 1, 2).setValues([[pkBaru, m ? m[1] : dP[i][8]]]);
            shP.getRange(i + 1, 16).setValue(new Date());
            break;
          }
        }
      } catch (e2) {}
    }
    const jabBaru = String(p.jabatanBaru || '').trim();
    const unitBaru = String(p.unitKerjaBaru || '').trim();
    if (!jabBaru) return;

    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.PEGAWAI);
    const digits = function (v) { return String(v || '').replace(/\D/g, ''); };
    let hit = -1;
    let data = [];
    // jalur cepat: cari baris pegawai lewat ID (satu kolom) lalu baca satu baris saja
    if (p.pegawaiID) {
      const lp = sheet.getLastRow();
      if (lp >= 2) {
        const f = sheet.getRange(2, 1, lp - 1, 1).createTextFinder(String(p.pegawaiID)).matchEntireCell(true).findNext();
        if (f) {
          hit = f.getRow() - 1;
          data[hit] = sheet.getRange(f.getRow(), 1, 1, Math.min(22, sheet.getMaxColumns())).getValues()[0];
        }
      }
    }
    if (hit < 0) {
      data = sheet.getDataRange().getValues();
      const k = digits(p.nip);
      if (k.length >= 8) for (let i = 1; i < data.length; i++) { if (digits(data[i][5]) === k || digits(data[i][6]) === k) { hit = i; break; } }
    }
    if (hit < 0) return;
    const row = data[hit];
    while (row.length < 22) row.push('');
    const tmt = String(p.tmt || '').substring(0, 10);

    if (isPlt) {
      const jenis = nm.indexOf('harian') !== -1 || /\bplh\b/.test(nm) ? 'PLH' : 'PLT';
      const oldTmt = row[19] ? formatDateString(row[19]) : '';
      if (oldTmt && tmt && tmt < oldTmt && String(row[21]) !== String(kegID)) return;
      if (row[16] && (String(row[16]) !== jabBaru || String(row[21]) !== String(kegID))) {
        logRiwayatJabatan_(auth, row, kegID, p.namaJenisKegiatan, String(row[18] || 'PLT'), row[16], row[17], '', jabBaru, unitBaru, tmt);
      }
      sheet.getRange(hit + 1, 17, 1, 6).setValues([[jabBaru, unitBaru, jenis, tmt, row[20], kegID]]);
      sheet.getRange(hit + 1, 16).setValue(new Date());
      return;
    }

    const oldTmtJ = row[20] ? formatDateString(row[20]) : '';
    if (oldTmtJ && tmt && tmt < oldTmtJ) return;
    let unor = row[12], induk = row[13];
    if (unitBaru) {
      const pos = unitBaru.lastIndexOf(' - ');
      if (pos > 0) { unor = unitBaru.substring(0, pos).trim(); induk = unitBaru.substring(pos + 3).trim(); }
      else { unor = unitBaru; }
    }
    if (String(row[9]) !== jabBaru || String(row[12]) !== String(unor) || String(row[13]) !== String(induk)) {
      logRiwayatJabatan_(auth, row, kegID, p.namaJenisKegiatan, 'DEFINITIF', row[9], row[12], row[13], jabBaru, unitBaru, tmt);
      sheet.getRange(hit + 1, 10).setValue(jabBaru);
      sheet.getRange(hit + 1, 13, 1, 2).setValues([[unor, induk]]);
    }
    sheet.getRange(hit + 1, 21).setValue(tmt);
    sheet.getRange(hit + 1, 16).setValue(new Date());
  } catch (e) {}
}

function clearPltPegawai_(auth, kegID, alasan) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.PEGAWAI);
    const last = sheet.getLastRow();
    if (last < 2 || sheet.getLastColumn() < 22) return;
    const f = sheet.getRange(2, 22, last - 1, 1).createTextFinder(String(kegID)).matchEntireCell(true).findNext();
    if (!f) return;
    const r = f.getRow();
    const row = sheet.getRange(r, 1, 1, 22).getValues()[0];
    logRiwayatJabatan_(auth, row, kegID, alasan || 'PLT/PLH berakhir', String(row[18] || 'PLT'), row[16], row[17], '', '', '', row[19] ? formatDateString(row[19]) : '');
    sheet.getRange(r, 17, 1, 3).setValues([['', '', '']]);
    sheet.getRange(r, 20).setValue('');
    sheet.getRange(r, 22).setValue('');
    bustData_();
  } catch (e) {}
}

function getRiwayatJabatanPegawai(token, pegawaiID) {
  requireAuth(token);
  ensureDatabaseStructure();
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAMES.RIWAYAT_JABATAN);
  const last = sh.getLastRow();
  if (last < 2) return [];
  const rows = sh.getRange(2, 1, last - 1, 16).getValues();
  const out = [];
  for (let i = rows.length - 1; i >= 0 && out.length < 200; i--) {
    const r = rows[i];
    if (pegawaiID && String(r[2]) !== String(pegawaiID)) continue;
    out.push({ logID: r[0], waktu: r[1] ? formatDateTimeString(r[1]) : '', pegawaiID: r[2], nama: r[3], kegiatanID: r[4], namaJenisKegiatan: r[5], jenis: r[6],
      jabatanSebelum: r[7], unorSebelum: r[8], unitIndukSebelum: r[9], jabatanSesudah: r[10], unorSesudah: r[11], tmt: r[12], username: r[13], pangkatSebelum: r[14] || '', pangkatSesudah: r[15] || '' });
  }
  return out;
}
